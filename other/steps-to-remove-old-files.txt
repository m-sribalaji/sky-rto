$runKey = "HKCU:\Software\Microsoft\Windows\CurrentVersion\Run"

# remove any startup shortcut if present
$startup = [Environment]::GetFolderPath("Startup")
Remove-Item -Force -ErrorAction SilentlyContinue (Join-Path $startup "SkyRTOTracker.lnk")

$base = Join-Path $env:USERPROFILE ".rto_tracker"
